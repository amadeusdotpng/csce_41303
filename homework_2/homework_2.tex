\begin{listing}[h]
\centering
\begin{minted}[linenos]{python}
def insert_sort(A, i=None):
    '''Sort A[:i+1]'''
    if i is None:
        i = len(A) - 1
    if i > 0:
        insertion_sort(A, i - 1)
        insert_last(A, i)

def insert_last(A, i):
    '''Sort A[:i+1] assuming sorted A[:i]'''
    if i > 0 and A[i] < A[i-1]:
        A[i], A[i-1] = A[i-1], A[i]
        insert_last(A, i-1)
\end{minted}
